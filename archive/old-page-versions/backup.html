'teacherId' : 'studentId'], name: userData.name, timestamp: new Date().toISOString(), role: isTeacher ? "Teacher" : "Student" }).then(() => { console.log(`${isTeacher ? "Teacher" : "Student"} ID not found.`); } }).catch(error => { showErrorMessage('An error occurred.
